---
title: Order List
api: GET /api/tron/gas/record/list
description: "Retrieve specific order records through ID"
---

#### Request Parameters

<ParamField query="request_ids" type="string" required>
  Comma-separated `request_id` strings
</ParamField>

#### Response Parameters

<Expandable title="object">
 <ResponseField name="id" type="integer">
     Order ID
  </ResponseField>
  <ResponseField name="trade_no" type="string">
    Resource purchase transaction No
  </ResponseField>
  <ResponseField name="request_id" type="string">
    Unique transaction identifier
  </ResponseField>
  <ResponseField name="status" type="integer">
    Status: `0` Order creation successful, `1` Proxy resource successful, `2` Proxy resource failed, `3` partially successful, `10` Recycling successful
  </ResponseField>
  <ResponseField name="service_charge_type" type="string">
    Service charge type. Energy and bandwidth flash recharge both use `10000`.
  </ResponseField>
  <ResponseField name="receive_address" type="string">
    resource receive address
  </ResponseField>
  <ResponseField name="amount" type="decimal">
    Payment amount. For flash recharge orders this is an estimate based on the purchased quantity until settlement completes, after which it is updated to the final settlement amount.
  </ResponseField>
  <ResponseField name="energy_num" type="integer">
    energy num
  </ResponseField>
  <ResponseField name="delegate_energy_num" type="integer">
    delegate energy num
  </ResponseField>
  <ResponseField name="reclaim_energy_num" type="integer">
    reclaim energy num
  </ResponseField>
    <ResponseField name="net_num" type="integer">
    net num
  </ResponseField>
     <ResponseField name="order_source" type="integer">
    order source: `0` Manual order, `1` API purchase, `2` Automatic lease, `3` Energy sublease
  </ResponseField>
   <ResponseField name="delegate_net_num" type="integer">
       delegate net num
  </ResponseField>
    <ResponseField name="reclaim_net_num" type="integer">
    reclaim net nu
  </ResponseField>
  <ResponseField name="item_list" type="string">
    energy sub order
    <Expandable title="object">
      <ResponseField name="energy_num" type="integer">
        energy num
      </ResponseField>
      <ResponseField name="energy_price" type="string">
        energy price,unit SUN

      <ResponseField name="energy_txid" type="string">
        energy txid
      </ResponseField>
      <ResponseField name="delegate_time" type="string">
        delegate time
      </ResponseField>
      <ResponseField name="resource_address" type="string">
        resource address
      </ResponseField>
      <ResponseField name="reclaim_txid" type="string">
        reclaim txid
      </ResponseField>
      <ResponseField name="reclaim_time" type="string">
        reclaim time
      </ResponseField>
    </Expandable>
  </ResponseField>
  <ResponseField name="net_item_list" type="string">
      net sub order
    <Expandable title="object">
      <ResponseField name="net_num" type="integer">
        net num
      </ResponseField>
      <ResponseField name="net_price" type="string">
        net price,unit SUN
      </ResponseField>
      <ResponseField name="net_txid" type="string">
        net txid

      <ResponseField name="delegate_time" type="string">
        delegate time

      <ResponseField name="resource_address" type="string">
        resource address

      <ResponseField name="reclaim_txid" type="string">
       net reclaim txid
      </ResponseField>
      <ResponseField name="reclaim_time" type="string">
        net reclaim time
      </ResponseField>
    </Expandable>
  </ResponseField>
  <ResponseField name="swap_info" type="object">
    Energy or bandwidth flash recharge information. This field is returned only for flash recharge orders and is omitted for regular orders.
    <Expandable title="object">
      <ResponseField name="address" type="string">Resource receiving address</ResponseField>
      <ResponseField name="energy_used" type="integer">Energy used before flash recharge</ResponseField>
      <ResponseField name="energy_num" type="integer">Available energy before flash recharge</ResponseField>
      <ResponseField name="buy_energy_num" type="integer">Energy quantity originally purchased by the user</ResponseField>
      <ResponseField name="send_energy_num" type="integer">Energy actually delegated</ResponseField>
      <ResponseField name="end_energy_used" type="integer">Energy used after reclaim is complete</ResponseField>
      <ResponseField name="end_energy_num" type="integer">Available energy after reclaim is complete</ResponseField>
      <ResponseField name="freeze_energy_num" type="integer">Energy base used for the initial lock</ResponseField>
      <ResponseField name="net_used" type="integer">Staked bandwidth used before recharge, excluding free bandwidth</ResponseField>
      <ResponseField name="net_num" type="integer">Staked bandwidth available before recharge, excluding free bandwidth; not the purchased quantity</ResponseField>
      <ResponseField name="buy_net_num" type="integer">Bandwidth quantity originally purchased by the user</ResponseField>
      <ResponseField name="send_net_num" type="integer">Bandwidth actually delegated; not the final billed quantity</ResponseField>
      <ResponseField name="end_net_used" type="integer">Staked bandwidth used after reclaim, excluding free bandwidth</ResponseField>
      <ResponseField name="end_net_num" type="integer">Staked bandwidth available after reclaim, excluding free bandwidth</ResponseField>
      <ResponseField name="freeze_net_num" type="integer">Bandwidth base used for the initial fund lock; not the final billed quantity</ResponseField>
      <ResponseField name="freeze_amount" type="string">Initially locked amount in TRX</ResponseField>
      <ResponseField name="current_lock_amount" type="string">Current locked amount in TRX</ResponseField>
      <ResponseField name="settlement_amount" type="string">Final settlement amount in TRX, calculated from the final billed resource quantity confirmed by the system after reclaim, using the price and fee-rate snapshot taken at order time</ResponseField>
    </Expandable>
  </ResponseField>
</Expandable>
